cmake_minimum_required(VERSION 3.20)
project(AnyDropShell LANGUAGES CXX)
add_library(AnyDropShell SHARED AnyDropShell.cpp exports.def)
target_compile_features(AnyDropShell PRIVATE cxx_std_17)
target_compile_definitions(AnyDropShell PRIVATE UNICODE _UNICODE)
target_compile_options(AnyDropShell PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(AnyDropShell PRIVATE shlwapi shell32 ole32 advapi32 user32 uuid)
set_property(TARGET AnyDropShell PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(AnyDropShellTest shell_test.cpp)
target_compile_features(AnyDropShellTest PRIVATE cxx_std_17)
target_compile_options(AnyDropShellTest PRIVATE /utf-8 /UNDEBUG)
target_link_libraries(AnyDropShellTest PRIVATE shell32 ole32 uuid)
enable_testing()
add_test(NAME native-shell-commands COMMAND AnyDropShellTest $<TARGET_FILE:AnyDropShell>)
